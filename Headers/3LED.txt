#include <avr/io.h>
#include "std_type.h"
#include "bit_math.h"
#include "DIO.h"

int main(void)
{
	DIO_voidSetPinDirection(DIO_PORTA,DIO_PIN0,DIO_INPUT);
	DIO_voidSetPinDirection(DIO_PORTA,DIO_PIN1,DIO_INPUT);
	DIO_voidSetPinDirection(DIO_PORTA,DIO_PIN2,DIO_INPUT);
	DIO_voidSetPinDirection(DIO_PORTA,DIO_PIN7,DIO_INPUT);
	
	DIO_voidSetPinDirection(DIO_PORTC,DIO_PIN0,DIO_OUTPUT);
	DIO_voidSetPinDirection(DIO_PORTC,DIO_PIN1,DIO_OUTPUT);
	DIO_voidSetPinDirection(DIO_PORTC,DIO_PIN2,DIO_OUTPUT);

	DIO_voidSetPinValue( DIO_PORTA, DIO_PIN0 ,0);
	DIO_voidSetPinValue( DIO_PORTA, DIO_PIN1 ,0);
	DIO_voidSetPinValue( DIO_PORTA, DIO_PIN2 ,0);
	DIO_voidSetPinValue( DIO_PORTA, DIO_PIN3 ,0);
	
    /* Replace with your application code */
    while (1) 
    {
		u8 A0 = DIO_u8GetPinValue(DIO_PORTA,DIO_PIN0);
		u8 A1 = DIO_u8GetPinValue(DIO_PORTA,DIO_PIN1);
		u8 A2 = DIO_u8GetPinValue(DIO_PORTA,DIO_PIN2);
		u8 A7 = DIO_u8GetPinValue(DIO_PORTA,DIO_PIN7);
		
		if (A0 == 0) {
			DIO_voidSetPinValue( DIO_PORTC, DIO_PIN0 ,1);
		} 
		else if(A1 == 0){
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN1 , 1);
	    }
		else if(A2 == 0){
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN2 , 1);
		}
		else if(A7 == 0){
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN0 , 1);
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN1 , 1);
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN2 , 1);
		}
		else{
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN0 , 0);
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN1 , 0);
			DIO_voidSetPinValue(DIO_PORTC, DIO_PIN2 , 0);
		}
    }
}
